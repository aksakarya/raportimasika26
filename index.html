// Ganti dengan ID Spreadsheet kamu (ambil dari URL Google Sheets) const SPREADSHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId(); function doPost(e) { try { const request = JSON.parse(e.postData.contents); const action = request.action; const payload = request.payload; let responseData = {}; switch (action) { case 'login': responseData = handleLogin(payload.username, payload.password); break; case 'getDashboardData': responseData = handleGetDashboardData(payload.userId, payload.role, payload.divisi, payload.bulan); break; case 'savePenilaian': responseData = handleSavePenilaian(payload); break; case 'toggleLock': responseData = handleToggleLock(payload); break; case 'saveFeedback': responseData = handleSaveFeedback(payload); break; default: throw new Error('Aksi tidak valid atau belum diimplementasi.'); } return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: responseData })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.message })) .setMimeType(ContentService.MimeType.JSON); } } function handleLogin(username, password) { const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName('Users'); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { const [userId, dbUser, dbPassHash, nama, divisi, role] = data[i]; // Normalisasi data: ubah ke format string dan hapus spasi berlebih const cleanDbUser = String(dbUser || '').trim(); const cleanDbPass = String(dbPassHash || '').trim(); const cleanInputUser = String(username || '').trim(); const cleanInputPass = String(password || '').trim(); const hashedInput = computeHash(cleanInputPass); // Pengecekan mendukung teks biasa (plain text) untuk testing if (cleanDbUser === cleanInputUser && (cleanDbPass === hashedInput || cleanDbPass === cleanInputPass)) { return { userId, nama, divisi, role }; } } throw new Error('Username atau password salah.'); } function handleGetDashboardData(userId, role, divisi, bulan) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const result = {}; const safeStr = (val) => String(val || '').trim(); const targetBulan = safeStr(bulan); const targetUserId = safeStr(userId); const targetDivisi = safeStr(divisi); if (role === 'Staf') { const logSheet = ss.getSheetByName('Penilaian_Staf'); if (!logSheet) return { myScores: [] }; const logs = logSheet.getDataRange().getValues(); // Filter kebal tipe data (ubah semua jadi string) result.myScores = logs.filter(row => safeStr(row[1]) === targetBulan && safeStr(row[2]) === targetUserId); const feedbackSheet = ss.getSheetByName('Feedback_BPH'); if (feedbackSheet) { const feedbacks = feedbackSheet.getDataRange().getValues(); result.feedback = feedbacks.find(row => safeStr(row[1]) === targetBulan && safeStr(row[2]) === targetDivisi); } } else if (role === 'Kadiv') { const usersSheet = ss.getSheetByName('Users'); const usersData = usersSheet.getDataRange().getValues(); result.staffList = usersData.filter(row => safeStr(row[4]) === targetDivisi && safeStr(row[5]) === 'Staf'); const configSheet = ss.getSheetByName('Divisi_Config'); if(configSheet) { const configData = configSheet.getDataRange().getValues(); result.config = configData.find(row => safeStr(row[1]) === targetBulan && safeStr(row[2]) === targetDivisi); } const logSheet = ss.getSheetByName('Penilaian_Staf'); if(logSheet) { const logs = logSheet.getDataRange().getValues(); const staffIds = result.staffList.map(s => safeStr(s[0])); result.scoreList = logs.filter(row => safeStr(row[1]) === targetBulan && staffIds.includes(safeStr(row[2]))); } } else if (role === 'Admin') { const logSheet = ss.getSheetByName('Penilaian_Staf'); if(logSheet) result.allScores = logSheet.getDataRange().getValues(); const configSheet = ss.getSheetByName('Divisi_Config'); if(configSheet) result.allConfigs = configSheet.getDataRange().getValues(); const feedbackSheet = ss.getSheetByName('Feedback_BPH'); if (feedbackSheet) result.allFeedback = feedbackSheet.getDataRange().getValues(); } return result; } function handleSavePenilaian(payload) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const safeStr = (val) => String(val || '').trim(); const targetBulan = safeStr(payload.bulan); const targetDivisi = safeStr(payload.divisi); // Cek apakah bulan ini sudah dikunci (Locked) const configSheet = ss.getSheetByName('Divisi_Config'); let configData = configSheet.getDataRange().getValues(); let configIndex = configData.findIndex(row => safeStr(row[1]) === targetBulan && safeStr(row[2]) === targetDivisi); if (configIndex > 0) { configSheet.getRange(configIndex + 1, 4).setValue(payload.totalRapat); configSheet.getRange(configIndex + 1, 5).setValue(payload.totalProker); } else { configSheet.appendRow([`C-${targetBulan}-${targetDivisi}`, payload.bulan, payload.divisi, payload.totalRapat, payload.totalProker, false]); } // Simpan Nilai Staf const logSheet = ss.getSheetByName('Penilaian_Staf'); const data = logSheet.getDataRange().getValues(); let savedCount = 0; payload.scores.forEach(scoreData => { const targetUserId = safeStr(scoreData.userId); const existingRowIndex = data.findIndex(row => safeStr(row[1]) === targetBulan && safeStr(row[2]) === targetUserId); const rowData = [ `L-${payload.bulan}-${scoreData.userId}`, payload.bulan, scoreData.userId, scoreData.hadirRapat, scoreData.lateRapat, scoreData.hadirProker, scoreData.lateProker, scoreData.teamwork, scoreData.komunikasi, scoreData.inisiatif, scoreData.keaktifan, scoreData.totalNilai, scoreData.predikat, scoreData.namaStaf, payload.divisi ]; if (existingRowIndex > 0) { logSheet.getRange(existingRowIndex + 1, 1, 1, rowData.length).setValues([rowData]); } else { logSheet.appendRow(rowData); } savedCount++; }); return { message: `Berhasil menyimpan data ${savedCount} staf.` }; } function handleToggleLock(payload) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const configSheet = ss.getSheetByName('Divisi_Config'); const data = configSheet.getDataRange().getValues(); const rowIndex = data.findIndex(row => row[1] === payload.bulan && row[2] === payload.divisi); if (rowIndex > 0) { configSheet.getRange(rowIndex + 1, 6).setValue(payload.isLocked); return { message: `Status kuncian Divisi ${payload.divisi} berhasil diperbarui.` }; } else { // Buat config baru jika belum ada configSheet.appendRow([`C-${payload.bulan}-${payload.divisi}`, payload.bulan, payload.divisi, 0, 0, payload.isLocked]); return { message: `Dibuat dan status kuncian diperbarui.` }; } } function handleSaveFeedback(payload) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let feedbackSheet = ss.getSheetByName('Feedback_BPH'); if (!feedbackSheet) { feedbackSheet = ss.insertSheet('Feedback_BPH'); feedbackSheet.appendRow(['FeedbackID', 'Bulan', 'Divisi', 'UserID_BPH', 'CatatanFeedback']); } const data = feedbackSheet.getDataRange().getValues(); const rowIndex = data.findIndex(row => row[1] === payload.bulan && row[2] === payload.divisi); const rowData = [`FB-${payload.bulan}-${payload.divisi}`, payload.bulan, payload.divisi, payload.adminId, payload.catatan]; if (rowIndex > 0) { feedbackSheet.getRange(rowIndex + 1, 1, 1, rowData.length).setValues([rowData]); } else { feedbackSheet.appendRow(rowData); } return { message: 'Feedback berhasil disimpan.' }; } function computeHash(input) { const rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input, Utilities.Charset.UTF_8); return rawHash.map(byte => (byte < 0 ? byte + 256 : byte).toString(16).padStart(2, '0')).join(''); }
